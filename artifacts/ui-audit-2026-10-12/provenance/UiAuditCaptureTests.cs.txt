// Temporary audit capture driver. Uses only synthetic data and existing render infrastructure.
[TestClass]
public sealed class UiAuditCaptureTests
{
    private static void Capture(MainWindow window, string name, int width)
    {
        WpfTestHost.CompleteLayout(window);
        WpfTestHost.Capture(window, $"audit-{name}-{width}");
        Console.WriteLine($"{name}: actual {window.ActualWidth}x{window.ActualHeight} DIP; scale {VisualTreeHelper.GetDpi(window).DpiScaleX * 100}%");
    }
    private static void Finish(Task task)
    {
        var end = DateTime.UtcNow.AddSeconds(15);
        while (!task.IsCompleted && DateTime.UtcNow < end) WpfTestHost.PumpDispatcher();
        task.GetAwaiter().GetResult();
    }
    private static void Bottom(DependencyObject view)
    {
        foreach (var scroll in WpfTestHost.Descendants<ScrollViewer>(view).Where(s => s.IsVisible && s.ScrollableHeight > 0).ToArray()) scroll.ScrollToBottom();
    }

    [TestMethod]
    [DataRow(1920,1080)] [DataRow(1338,753)] [DataRow(720,560)]
    public void AuthenticationStates(int width,int height)
    {
        WpfTestHost.Run(width,height,w => {
            Capture(w,"sign-in",width);
            w.SignInSurface.ShowConnectionFailure(); Capture(w,"sign-in-failure",width);
            Bottom(w.SignInSurface); Capture(w,"sign-in-failure-bottom",width);
            w.SignInSurface.Reset();
            var pending = new TaskCompletionSource<Features.Authentication.AuthenticatedSurface>();
            w.SignInSurface.AuthenticateAsync = (_,_,_) => pending.Task;
            WpfTestHost.FindByName<TextBox>(w,"UsernameBox").Text="synthetic.user";
            WpfTestHost.FindByName<PasswordBox>(w,"PasswordInput").Password="synthetic-password";
            WpfTestHost.FindByName<Button>(w,"SignInButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            Capture(w,"sign-in-busy",width);
            pending.SetException(new System.IO.IOException("synthetic unavailable")); WpfTestHost.PumpDispatcher();
        },showSignIn:true);
    }

    [TestMethod]
    [DataRow(1920,1080)] [DataRow(1338,753)] [DataRow(720,560)]
    public void UserStates(int width,int height)
    {
        var engine = new FakeEngine();
        engine.DesktopAccounts.Add(new DesktopAccountSummary { AccountId=Guid.NewGuid(),UserId=Guid.NewGuid(),DisplayName="مستخدم تجريبي طويل الاسم",Username="synthetic.user",Role=DesktopAccountRole.Receptionist,Active=true,Revision=1 });
        WpfTestHost.Run(width,height,w=>{
            WpfTestHost.FindByName<Button>(w,"UsersNavButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            var v=WpfTestHost.Descendants<UsersView>(w).Single(); Finish(v.ViewModel.LoadAsync());
            Capture(w,"users-list",width);
            WpfTestHost.FindByAutomationName<Button>(v,"تعطيل المستخدم مستخدم تجريبي طويل الاسم").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            Capture(w,"users-deactivate",width);
            WpfTestHost.FindByName<Button>(v,"CancelDeactivateButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            v.ViewModel.BeginEdit(); Capture(w,"users-create",width);
            Bottom(v); Capture(w,"users-create-bottom",width);
            v.ViewModel.EditorName="مستخدم تجريبي"; v.ViewModel.EditorUsername="synthetic.user";
            Finish(v.ViewModel.ApplyAsync("short"));
            WpfTestHost.FindByName<TextBox>(v,"UserNameInput").BringIntoView(); Capture(w,"users-validation",width);
            v.ViewModel.CancelEditor(); v.ViewModel.SearchText="لا يوجد"; Capture(w,"users-empty",width);
            v.ViewModel.SearchText="";
            engine.QueryHandler = _=>throw new System.IO.IOException("synthetic unavailable");
            Finish(v.ViewModel.LoadAsync()); Capture(w,"users-unavailable",width);
        },engine:engine);
    }

    [TestMethod]
    [DataRow(1920,1080)] [DataRow(1338,753)] [DataRow(720,560)]
    public void FurnitureSteps(int width,int height)
    {
        var f=new Furniture.FurnitureFixtures(); f.Seed(); var engine=f.Engine();
        WpfTestHost.Run(width,height,w=>{
            WpfTestHost.FindByName<Button>(w,"FurnitureNavButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            WpfTestHost.CompleteLayout(w);
            var v=WpfTestHost.Descendants<FurnitureView>(w).Single();
            Finish(v.PrepareEditorAsync(v.ViewModel.VisibleFurniture.Single()));
            v.ViewModel.BeginEdit(v.ViewModel.VisibleFurniture.Single());
            for(var step=1;step<=6;step++) {
                Capture(w,$"furniture-step-{step}",width); Bottom(v); Capture(w,$"furniture-step-{step}-bottom",width);
                if(step<6) WpfTestHost.FindByName<Button>(v,"NextButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            }
            v.ViewModel.BeginEditVariant(v.ViewModel.Variants.Single()); Capture(w,"furniture-variant-dialog",width);
        },engine:engine);
    }

    [TestMethod]
    [DataRow(1920,1080)] [DataRow(1338,753)] [DataRow(720,560)]
    public void HomeSearchAndFailure(int width,int height)
    {
        var data=Home.HomeStateTests.Data();
        data.Customers=Home.HomeStateTests.Section(1,new HomeItem { Id=Guid.NewGuid(),Destination=HomeDestination.Customer,Title="عميل بحث تجريبي",State="نشط",ChangedAt=1791626400000 });
        var engine=Home.HomeStateTests.Engine(data);
        WpfTestHost.Run(width,height,w=>{
            var model=new HomeViewModel(engine); Finish(model.ActivateAsync()); w.ManagerHome.DataContext=model;
            Capture(w,"manager-home-top",width);
            model.Search("عميل"); Finish(model.LastLoad); Bottom(w.ManagerHome.Parent); Capture(w,"home-search",width);
            data.Quotations.Availability=HomeAvailability.Denied; data.Orders.ServerAvailable=false;
            model.Refresh(); Finish(model.LastLoad); Capture(w,"home-partial-denied",width);
            engine.QueryHandler = _=>throw new System.IO.IOException("synthetic unavailable");
            model.Refresh(); Finish(model.LastLoad); Capture(w,"home-unavailable",width);
            Finish(model.DisposeAsync().AsTask());
        });
    }

    }
