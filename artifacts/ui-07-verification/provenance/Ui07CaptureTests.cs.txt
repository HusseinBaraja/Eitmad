using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using Eitmad.Contracts;
using Eitmad.WindowsShell.Controls;
using Eitmad.WindowsShell.Tests.Products;
using Eitmad.WindowsShell.Tests.TestDoubles;

namespace Eitmad.WindowsShell.Tests.Rendered;

// Temporary visual verification driver; reuse the existing synthetic Products fixture.
[TestClass]
public sealed class Ui07CaptureTests
{
    [TestMethod]
    [DataRow(1536, 816)]
    [DataRow(1338, 753)]
    [DataRow(720, 560)]
    public void ManagerNavigation(int width, int height)
    {
        var data = ProductsPresentationTests.Data();
        var engine = new FakeEngine();
        engine.QueryHandler = query => new QueryResponseEnvelope
        {
            RequestId = Guid.NewGuid(), CorrelationId = Guid.NewGuid(),
            Outcome = new QueryOutcome
            {
                Status = CommandOutcomeStatus.Succeeded,
                Payload = query.Kind == Query.ProductCategoryListKind
                    ? QueryResult.ForProductCategories(data.Categories)
                    : QueryResult.ForProducts(new ProductPage { Items = data.Products.ToArray(), CanManage = true, CanReadCosts = true }),
            },
        };
        WpfTestHost.Run(width, height, window =>
        {
            window.Left = 0;
            window.Top = 0;
            WpfTestHost.FindByName<Button>(window, "ProductsNavButton").RaiseEvent(new RoutedEventArgs(Button.ClickEvent));
            WpfTestHost.CompleteLayout(window);
            var sidebar = WpfTestHost.FindByName<ShellSidebar>(window, "ManagerSidebar");
            var buttons = WpfTestHost.Descendants<Button>(sidebar).ToArray();
            Assert.AreEqual(10, buttons.Length);
            Assert.IsTrue(buttons.All(button => button.IsEnabled));
            Assert.IsTrue(window.ProductsSurface.IsVisible);
            var scroll = WpfTestHost.Descendants<ScrollViewer>(sidebar).Single();
            Console.WriteLine($"UI-07: window {window.ActualWidth}x{window.ActualHeight} DIP; scaling {VisualTreeHelper.GetDpi(window).DpiScaleX * 100}%; sidebar scroll range {scroll.ScrollableHeight} DIP; routes {string.Join(", ", buttons.Select(button => button.Tag))}");
            WpfTestHost.Capture(window, $"ui-07-products-{width}x{height}");
            if (scroll.ScrollableHeight > 0)
            {
                scroll.ScrollToBottom();
                WpfTestHost.CompleteLayout(window);
                var users = WpfTestHost.FindByName<Button>(sidebar, "UsersNavButton");
                var bounds = users.TransformToAncestor(scroll).TransformBounds(new Rect(users.RenderSize));
                Assert.IsTrue(bounds.Top >= 0 && bounds.Bottom <= scroll.ViewportHeight);
                WpfTestHost.Capture(window, $"ui-07-sidebar-bottom-{width}x{height}");
            }
        }, engine: engine);
    }
}
